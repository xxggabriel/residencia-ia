"""Assemble a distribution inventory without changing the scientific export manifest."""
import argparse
import hashlib
import json
from datetime import datetime, timezone
from pathlib import Path

EXPORT_SHA = '90929df482160a0d581d32d6e3a1438eb6e636eeac98a2f417f6576e7e953636'
FORBIDDEN = {'.db', '.sqlite', '.sqlite3', '.pkl', '.pickle', '.bin', '.ubj', '.pt', '.pth', '.ckpt', '.onnx', '.npy', '.npz', '.joblib'}


def digest(path):
    with path.open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()


def inventory(root, exclude=()):
    result = []
    for path in sorted(root.rglob('*')):
        if path.is_symlink():
            raise ValueError('A distributed artifact cannot be a symlink')
        if not path.is_file():
            continue
        name = path.relative_to(root).as_posix()
        if name in exclude:
            continue
        if path.suffix.lower() in FORBIDDEN or path.name in {'.DS_Store'} or path.suffix == '.tmp':
            raise ValueError(f'Unexpected artifact type: {name}')
        size = path.stat().st_size
        if size >= 100 * 1024**2:
            raise ValueError(f'Artifact exceeds the hosting limit: {name}')
        result.append({'path': name, 'bytes': size, 'sha256': digest(path)})
    return result


def write_sums(root, records):
    (root / 'SHA256SUMS').write_text(''.join(f"{record['sha256']}  {record['path']}\n" for record in records))


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--root', type=Path, required=True)
    parser.add_argument('--qa-reference', required=True)
    args = parser.parse_args()
    root = args.root.resolve()
    scientific = root / 'manifest.json'
    if digest(scientific) != EXPORT_SHA:
        raise ValueError('The original scientific manifest was modified')
    export = json.loads(scientific.read_text())
    native = json.loads((root / 'validation/mlflow/native-mlflow-verification-summary.json').read_text())
    qa = (root / args.qa_reference).resolve()
    if not qa.is_relative_to(root) or not qa.is_file():
        raise ValueError('The QA receipt must be included in the package')
    qa_result = json.loads(qa.read_text())
    if qa_result.get('complete') is not True or qa_result.get('failures') != []:
        raise ValueError('Approved export QA is required')
    if not export['complete'] or export['completed_cases'] != 60 or not native['complete']:
        raise ValueError('Completed scientific export and native verification are required')
    validation = root / 'validation'
    write_sums(validation, inventory(validation, exclude={'SHA256SUMS'}))
    records = inventory(root, exclude={'package-manifest.json', 'SHA256SUMS'})
    package = {
        'schema': 'corrective-evidence-distribution-v1',
        'assembled_at_utc': datetime.now(timezone.utc).isoformat(),
        'scientific_export': {'path': 'manifest.json', 'sha256': EXPORT_SHA, 'unchanged': True},
        'execution_revision': '98a6dd5a676f46bf1df337de82b33fd0e554c085',
        'exporter': {'revision': 'f43c7249559ad21d38ee2968502c61fdfed30a24', 'script_sha256': export['exporter']['script_sha256']},
        'execution_times_helper_revision': '07aa7f5acdd30101c446b733b134f82b172df352',
        'scope': {'matrix_cases': 60, 'replication_jobs': 8, 'replication_trials': 24, 'seeds': [42, 43, 44], 'online_cases_deferred': 12},
        'qa_reference': {'path': qa.relative_to(root).as_posix(), 'sha256': digest(qa)},
        'native_verification': {'path': 'validation/mlflow/native-mlflow-verification-summary.json', 'sha256': digest(root / 'validation/mlflow/native-mlflow-verification-summary.json')},
        'assembly_script_sha256': digest(Path(__file__)),
        'retention': export['retention'],
        'files': records,
        'indexed_bytes': sum(item['bytes'] for item in records),
        'excludes_from_file_inventory': ['package-manifest.json', 'SHA256SUMS'],
        'checksum_contract': 'SHA256SUMS includes package-manifest.json and every other distributed file except itself; manifest.json remains the original scientific export.'
    }
    (root / 'package-manifest.json').write_text(json.dumps(package, ensure_ascii=False, indent=2, sort_keys=True, allow_nan=False) + '\n')
    write_sums(root, inventory(root, exclude={'SHA256SUMS'}))
    assert digest(scientific) == EXPORT_SHA
    print(json.dumps({'files': len(records), 'indexed_bytes': package['indexed_bytes'], 'scientific_manifest_unchanged': True}))


if __name__ == '__main__':
    main()
