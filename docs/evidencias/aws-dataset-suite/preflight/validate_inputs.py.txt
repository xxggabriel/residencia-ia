"""Read every prepared row and record aggregate integrity checks without modifying inputs."""
import datetime
import hashlib
import json
from pathlib import Path

import numpy as np
import pyarrow.parquet as pq

root = Path('/prepared')
results = []
for directory in sorted(root.iterdir()):
    path = directory / 'full.parquet'
    manifest_path = directory / 'full.json'
    manifest = json.loads(manifest_path.read_text())
    classes = len(manifest['source']['classes'])
    counts = np.zeros(classes, dtype=np.int64)
    last = None
    rows = missing = required_missing = infinite = decreasing = duplicate_times = 0
    feature_missing = np.zeros(len(manifest['features']), dtype=np.int64)
    low = np.full(len(manifest['features']), np.inf)
    high = np.full(len(manifest['features']), -np.inf)
    with pq.ParquetFile(path) as source:
        for batch in source.iter_batches(batch_size=65536, use_threads=False):
            missing += sum(column.null_count for column in batch.columns)
            required_missing += batch[0].null_count + batch[1].null_count
            timestamps = batch[0].to_numpy()
            labels = batch[1].to_numpy()
            assert np.all((labels >= 0) & (labels < classes))
            if last is not None:
                decreasing += int(timestamps[0] < last)
                duplicate_times += int(timestamps[0] == last)
            decreasing += int(np.count_nonzero(np.diff(timestamps) < 0))
            duplicate_times += int(np.count_nonzero(np.diff(timestamps) == 0))
            last = int(timestamps[-1])
            counts += np.bincount(labels, minlength=classes)
            for index, column in enumerate(batch.columns[2:]):
                values = column.to_numpy(zero_copy_only=False)
                feature_missing[index] += int(np.count_nonzero(np.isnan(values)))
                infinite += int(np.count_nonzero(np.isinf(values)))
                low[index] = min(low[index], float(np.nanmin(values)))
                high[index] = max(high[index], float(np.nanmax(values)))
            rows += batch.num_rows
    with path.open('rb') as stream:
        digest = hashlib.file_digest(stream, 'sha256').hexdigest()
    assert rows == manifest['rows'] and digest == manifest['sha256']
    assert not required_missing and not infinite and not decreasing
    result = {
        'dataset': manifest['dataset'], 'scope': 'full', 'rows': rows,
        'prepared_sha256': digest,
        'manifest_sha256': hashlib.sha256(manifest_path.read_bytes()).hexdigest(),
        'class_counts': counts.tolist(), 'arrow_null_values': missing,
        'required_timestamp_or_label_nulls': required_missing,
        'feature_missing_counts': dict(zip(manifest['features'], feature_missing.tolist(), strict=True)),
        'infinite_features': infinite, 'decreasing_timestamps': decreasing,
        'feature_missing_policy': 'preserved; model-native missing handling and warmup-only geometry fill',
        'adjacent_timestamp_ties': duplicate_times,
        'feature_min': dict(zip(manifest['features'], low.tolist(), strict=True)),
        'feature_max': dict(zip(manifest['features'], high.tolist(), strict=True)),
        'complete': True,
    }
    results.append(result)
    print(json.dumps({'dataset': manifest['dataset'], 'rows': rows, 'complete': True}), flush=True)
Path('/evidence/input-integrity.json').write_text(json.dumps({
    'schema': 'dataset-suite-input-integrity-v1',
    'verified_at_utc': datetime.datetime.now(datetime.timezone.utc).isoformat(),
    'complete': len(results) == 7, 'sources': results,
}, indent=2, allow_nan=False) + '\n')
